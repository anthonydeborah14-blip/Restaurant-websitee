                          <img src={item.image} alt={item.name} className="w-12 h-12 rounded-lg object-cover" />
                          <div>
                            <h4 className="font-serif text-sm font-bold text-white">{item.name}</h4>
                            <span className="text-xs text-[#D4AF37]">₦{(item.price * item.quantity).toLocaleString()}</span>
                          </div>
                        </div>

                        <div className="flex items-center gap-2 bg-[#0A1C14] border border-[#D4AF37]/30 rounded-lg px-2 py-1">
                          <button onClick={() => updateCartQty(item.id, -1)} className="text-[#D4AF37] p-1"><Minus className="w-3 h-3" /></button>
                          <span className="text-xs font-bold text-white">{item.quantity}</span>
                          <button onClick={() => updateCartQty(item.id, 1)} className="text-[#D4AF37] p-1"><Plus className="w-3 h-3" /></button>
                        </div>
                      </div>
                    ))}
                  </div>

                  {/* Order Summary Pricing */}
                  <div className="bg-[#0A1C14] p-4 rounded-xl border border-[#D4AF37]/20 space-y-2 text-xs">
                    <div className="flex justify-between text-[#EFECE6]/70">
                      <span>Subtotal</span>
                      <span>₦{cartSubtotal.toLocaleString()}</span>
                    </div>
                    {deliveryType === 'delivery' && (
                      <div className="flex justify-between text-[#EFECE6]/70">
                        <span>Delivery Fee</span>
                        <span>₦{deliveryFee.toLocaleString()}</span>
                      </div>
                    )}
                    <div className="flex justify-between font-bold text-[#D4AF37] pt-2 border-t border-[#D4AF37]/10 text-sm">
                      <span>Total Amount</span>
                      <span>₦{cartTotal.toLocaleString()}</span>
                    </div>
                  </div>
                </>
              )}

            </div>

            {/* Footer Checkout CTA */}
            {cart.length > 0 && !orderConfirmedCode && (
              <div className="p-6 bg-[#07130E] border-t border-[#D4AF37]/20">
                <button 
                  onClick={handleCheckout}
                  disabled={isCheckingOut}
                  className="w-full py-3.5 bg-[#D4AF37] hover:bg-[#B8860B] text-[#0A1C14] font-bold text-sm rounded-xl shadow-lg transition flex items-center justify-center gap-2"
                >
                  {isCheckingOut ? (
                    <span className="animate-pulse">Processing Order...</span>
                  ) : (
                    <>
                      <span>Checkout & Pay (₦{cartTotal.toLocaleString()})</span>
                      <ArrowRight className="w-4 h-4" />
                    </>
                  )}
                </button>
              </div>
            )}

          </div>
        </div>
      )}

      {/* --- MODAL 3: LIGHTBOX GALLERY --- */}
      {lightboxImage && (
        <div className="fixed inset-0 z-50 bg-black/90 flex items-center justify-center p-4">
          <div className="relative max-w-4xl w-full">
            <button 
              onClick={() => setLightboxImage(null)}
              className="absolute -top-12 right-0 text-white hover:text-[#D4AF37]"
            >
              <X className="w-8 h-8" />
            </button>
            <img src={lightboxImage.url} alt={lightboxImage.title} className="w-full max-h-[80vh] object-contain rounded-2xl" />
            <div className="text-center mt-4">
              <h3 className="font-serif text-xl font-bold text-white">{lightboxImage.title}</h3>
            </div>
          </div>
        </div>
      )}

      {/* FLOATING WHATSAPP BUTTON */}
      <a 
        href="https://wa.me/2348031234567" 
        target="_blank" 
        rel="noopener noreferrer"
        className="fixed bottom-6 right-6 z-30 bg-[#25D366] text-white p-3.5 rounded-full shadow-2xl hover:scale-110 transition duration-300 flex items-center justify-center"
        title="Chat with Reservations Concierge"
      >
        <MessageCircle className="w-6 h-6" />
      </a>

    </div>
  );
}
